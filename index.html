// 1. 自動朗讀功能 (TTS) function speak() { const text = document.getElementById('modal-text').innerText; window.speechSynthesis.cancel(); // 先停止之前的語音 const msg = new SpeechSynthesisUtterance(text); msg.lang = 'zh-HK'; // 設定為粵語 msg.rate = 0.8; // 速度稍微慢一點，方便學生聽清楚 window.speechSynthesis.speak(msg); } // 2. AI 語音辨識功能 (STT) function startRecognition() { const recognition = new (window.SpeechRecognition || window.webkitSpeechRecognition)(); recognition.lang = 'zh-HK'; const btnAi = document.getElementById('btn-ai'); btnAi.innerText = "👂 正在聽...請朗讀"; btnAi.style.background = "#ed64a6"; recognition.start(); recognition.onresult = (event) => { const result = event.results[0][0].transcript; const target = document.getElementById('modal-text').innerText; // 判斷是否包含目標詞語 if(result.includes(target) || target.includes(result)) { alert("✅ 太棒了！讀音正確 \n(識別到: " + result + ")"); closeModal(true); // 自動關閉並加分 } else { alert("❌ 好像不太對，再試試看？ \n(識別到: " + result + ")"); } resetAiBtn(); }; recognition.onerror = () => { alert("🎤 沒聽到聲音，請檢查麥克風權限。"); resetAiBtn(); }; function resetAiBtn() { btnAi.innerText = "🤖 AI 檢查讀音"; btnAi.style.background = "#4299e1"; } }
